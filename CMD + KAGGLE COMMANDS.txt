import os
import subprocess
from pathlib import Path

def run(cmd, cwd=None):
    print(f"\n>>> {cmd}")
    subprocess.run(cmd, shell=True, check=True, cwd=cwd)

p = Path(
    "/usr/local/lib/python3.12/dist-packages/"
    "basicsr/data/degradations.py"
)

if not p.exists():
    raise FileNotFoundError(f"BasicSR file not found: {p}")

text = p.read_text()

old = (
    "from torchvision.transforms.functional_tensor "
    "import rgb_to_grayscale"
)

new = (
    "from torchvision.transforms.functional "
    "import rgb_to_grayscale"
)

if old in text:
    p.write_text(text.replace(old, new))
    print("\nBasicSR PATCHED successfully.")
elif new in text:
    print("\nBasicSR already patched.")
else:
    raise RuntimeError(
        "Could not find expected rgb_to_grayscale import."
